<script>
const DB = {
  async call(e, d) { const r = await fetch(e, {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify(d)}); return await r.json(); },
  async verify(u, p) { return this.call('/api/login', {username:u, passwordHash:btoa(p)}); },
  async getRecs(id) { const r = await this.call('/api/records', {userId:id}); return r.records || []; },
  async addRec(id, r) { return this.call('/api/record/create', {userId:id, record:r}); },
  async delRec(id) { return this.call('/api/record/delete', {id:id}); },
  async genQuiz(id, t, txt) { return this.call('/api/quiz/generate', {userId:id, title:t, text:txt}); },
  async getQuizzes(id) { const r = await this.call('/api/quizzes', {userId:id}); return r.quizzes || []; }
};

let currentUser = null, allData = [], allQuizzes = [];

async function initApp(user) {
  currentUser = user; 
  allData = await DB.getRecs(user.id);
  allQuizzes = await DB.getQuizzes(user.id);
  document.getElementById('authSection').classList.add('hidden');
  document.getElementById('appSection').classList.remove('hidden');
  document.getElementById('userAvatar').textContent = user.fullName[0];
  renderAll();
}

function showLogin() { document.getElementById('loginPage').classList.remove('hidden'); document.getElementById('recoverPage').classList.add('hidden'); }
function showRecover() { document.getElementById('loginPage').classList.add('hidden'); document.getElementById('recoverPage').classList.remove('hidden'); }
function logout() { location.reload(); }

document.getElementById('loginForm').addEventListener('submit', async (e) => {
  e.preventDefault();
  const res = await DB.verify(document.getElementById('loginUsername').value, document.getElementById('loginPassword').value);
  if (res.ok) initApp(res.user); else alert(res.error);
});

function switchTab(tab) {
  document.querySelectorAll('section').forEach(s => s.classList.add('hidden'));
  document.getElementById('page-' + tab).classList.remove('hidden');
  document.querySelectorAll('nav button').forEach(b => b.style.opacity = '0.5');
  event.target.style.opacity = '1';
}

function readF(input) {
  const f = input.files[0]; if(!f) return;
  const r = new FileReader(); r.onload = e => document.getElementById('q_text').value = e.target.result;
  r.readAsText(f);
}

async function genQ() {
  const btn = document.getElementById('genBtn'); btn.disabled = true; btn.textContent = 'Жасалуда...';
  const res = await DB.genQuiz(currentUser.id, document.getElementById('q_title').value, document.getElementById('q_text').value);
  btn.disabled = false; btn.textContent = 'Генерациялау';
  if(res.ok) { allQuizzes = await DB.getQuizzes(currentUser.id); renderAll(); } else alert(res.error);
}

const GR_VAL = { 'A': 4.0, 'A-': 3.67, 'B+': 3.33, 'B': 3.0, 'B-': 2.67, 'C+': 2.33, 'C': 2.0, 'C-': 1.67, 'D+': 1.33, 'D': 1.0, 'F': 0 };

function renderAll() {
  const gpas = allData.filter(d => d.type === 'gpa');
  let pts = 0, crs = 0;
  document.getElementById('gpaList').innerHTML = gpas.map(g => {
    const weight = GR_VAL[g.grade] || 0; pts += weight * g.credits; crs += g.credits;
    return `<div class="app-card flex justify-between"><div>${g.subject} (${g.credits} кр)</div><div><b>${g.grade}</b></div></div>`;
  }).join('');
  const val = crs > 0 ? (pts / crs).toFixed(2) : "0.00";
  document.getElementById('gpaValue').textContent = val;
  document.getElementById('statGpa').textContent = val;
  
  document.getElementById('quizList').innerHTML = allQuizzes.map(q => `<div class="app-card">${q.title} - ${q.questions.length} сұрақ</div>`).join('');
}

function openModal(t) {
  const ov = document.getElementById('modalOverlay'); ov.classList.remove('hidden');
  if(t==='gpa') {
    ov.innerHTML = `<div class="modal-box space-y-3">
      <input id="ms" class="glass-input" placeholder="Пән атауы">
      <input id="mc" type="number" class="glass-input" placeholder="Кредит саны">
      <div class="grid grid-cols-3 gap-2">
        <input id="mr1" type="number" class="glass-input" placeholder="РК1">
        <input id="mr2" type="number" class="glass-input" placeholder="РК2">
        <input id="mex" type="number" class="glass-input" placeholder="Экз">
      </div>
      <button onclick="saveG()" class="btn-main">Сақтау</button>
    </div>`;
  }
}

async function saveG() {
  const r1 = parseFloat(document.getElementById('mr1').value)||0, r2 = parseFloat(document.getElementById('mr2').value)||0, ex = parseFloat(document.getElementById('mex').value)||0;
  const total = ((r1+r2)/2)*0.6 + (ex*0.4);
  let g = 'F';
  if(total>=95) g='A'; else if(total>=90) g='A-'; else if(total>=85) g='B+'; else if(total>=80) g='B'; else if(total>=75) g='B-'; else if(total>=70) g='C+'; else if(total>=65) g='C'; else if(total>=60) g='C-'; else if(total>=55) g='D+'; else if(total>=50) g='D';
  await DB.addRec(currentUser.id, {type:'gpa', subject:document.getElementById('ms').value, grade:g, credits:parseInt(document.getElementById('mc').value)});
  allData = await DB.getRecs(currentUser.id);
  document.getElementById('modalOverlay').classList.add('hidden'); renderAll();
}
